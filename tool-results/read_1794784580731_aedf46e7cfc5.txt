   260→export function AnalystPanel({ focusSymbol }: { focusSymbol: string }) {
   261→  // stable tz for the query string; bust counter forces a manual refetch by
   262→  // changing the url (the server cache key only reads `tz`, extra params are
   263→  // ignored) — this never changes on the 1s clock.
   264→  const [tz] = useState(() => localTzOffsetMin());
   265→  const [bust, setBust] = useState(0);
   266→  const url = `/api/analyst?tz=${tz}${bust > 0 ? `&_=${bust}` : ""}`;
   267→  const { data, error } = usePolling<AnalystResponse>(url, POLL_MS);
   268→
   269→  // 1s local clock — drives every visible countdown, never the fetch loop.
   270→  const [now, setNow] = useState(() => Date.now());
   271→  useEffect(() => {
   272→    const t = setInterval(() => setNow(Date.now()), 1000);
   273→    return () => clearInterval(t);
   274→  }, []);
   275→
   276→  // When the current payload first rendered — recomputed once per fetch (the
   277→  // data object identity changes on every response), so no effect is needed.
   278→  const receivedAt = useMemo(() => Date.now(), [data]);
   279→
   280→  // Manual refresh: bump the url (the polling effect refetches immediately)
   281→  // and remember the click time so the spinner stops as soon as a newer
   282→  // payload (receivedAt) or an error lands.
   283→  const [refreshClickedAt, setRefreshClickedAt] = useState(0);
   284→  const manualRefresh = () => {
   285→    setRefreshClickedAt(Date.now());
   286→    setBust((b) => b + 1);
   287→  };
   288→
   289→  const expired = error != null && error.toLowerCase().includes("session expired");
   290→
   291→  if (!data || expired) {
   292→    if (error) return <AnalystError error={error} onRetry={manualRefresh} />;
   293→    return <AnalystSkeleton />;
   294→  }
   295→
   296→  const refreshInSec = Math.max(0, Math.ceil((receivedAt + POLL_MS - now) / 1000));
   297→  const refreshingNow =
   298→    (refreshClickedAt > 0 && receivedAt <= refreshClickedAt && !error) || refreshInSec === 0;
   299→
   300→  return (
   301→    <section className="panel flex-1 min-h-0 flex flex-col overflow-hidden" aria-label="Market analyst">
   302→      {/* header bar */}
   303→      <header className="panel-head shrink-0 gap-2">
   304→        <div className="flex items-center gap-2 min-w-0">
   305→          <span className="truncate">AMI ANALYST</span>
   306→          <span className="chip hidden sm:inline-flex items-center gap-1.5">
   307→            <span className="w-1.5 h-1.5 rounded-full bg-[#4bc39a] pulse-dot" aria-hidden />
   308→            LIVE DESK BRIEFING
   309→          </span>
   310→          {error && (
   311→            <StatusPill tone="amber" title={error} className="hidden lg:inline-flex">
   312→              FEED RETRYING
   313→            </StatusPill>
   314→          )}
   315→        </div>
   316→        <div className="flex items-center gap-1.5 shrink-0">
   317→          <span className="chip hidden md:inline-flex tnum" title="Local time the current briefing was generated">
   318→            UPDATED {fmtTime(data.generatedAt)}
   319→          </span>
   320→          <span className="chip hidden sm:inline-flex tnum" title="Next automatic refresh of the analyst feed">
   321→            <span className="sr-only">Auto refresh countdown: </span>
   322→            {refreshingNow ? "REFRESHING…" : `REFRESH IN ${refreshInSec}S`}
   323→          </span>
   324→          <Tooltip>
   325→            <TooltipTrigger asChild>
   326→              <button type="button" className="focus:outline-none cursor-help" aria-label="Analyst model status">
   327→                <StatusPill tone={data.llm.used ? "green" : "amber"}>
   328→                  {data.llm.used
   329→                    ? `ANALYST MODEL · ${(data.llm.latencyMs / 1000).toFixed(1)}s`
   330→                    : "DETERMINISTIC FALLBACK"}
   331→                </StatusPill>
   332→              </button>
   333→            </TooltipTrigger>
   334→            <TooltipContent side="bottom" className="max-w-72 text-[10px] leading-snug break-words">
   335→              {data.llm.used
   336→                ? "Briefing written by the analyst model from the deterministic confluence context — every number traces back to live feeds."
   337→                : data.llm.note ?? "Analyst model unreachable — deterministic fallback briefing generated from the confluence engine."}
   338→            </TooltipContent>
   339→          </Tooltip>
   340→          <Button
   341→            variant="ghost"
   342→            size="sm"
   343→            onClick={manualRefresh}
   344→            aria-label="Refresh analyst feed now"
   345→            className="h-6 w-7 px-0 text-[#8193a3] hover:text-[#e5f0fb]"
   346→          >
   347→            <RefreshCw className={cn("w-3.5 h-3.5", refreshingNow && "animate-spin")} />
   348→          </Button>
   349→        </div>
   350→      </header>
   351→
   352→      {/* body */}
   353→      <div className="flex-1 min-h-0 overflow-y-auto ami-scroll p-2">
   354→        <div className="grid gap-2 grid-cols-1 xl:grid-cols-[minmax(0,1.1fr)_minmax(0,0.9fr)] items-start">
   355→          {/* left column — briefing feed, market pulse, engine signals */}
   356→          <div className="flex flex-col gap-2 min-w-0">
   357→            <BriefingFeed briefing={data.briefing} generatedAt={data.generatedAt} />
   358→            <PulseStrip pulses={data.pulses} focusSymbol={focusSymbol} />
   359→            <SignalsStrip signals={data.signals} />
   360→          </div>
   361→
   362→          {/* right column — sessions board, event risk board */}
   363→          <div className="flex flex-col gap-2 min-w-0">
   364→            <SessionsBoard sessions={data.sessions} now={now} />
   365→            <EventsBoard calendar={data.calendar} now={now} />
   366→          </div>
   367→
   368→          {/* full width — confluence board */}
   369→          <div className="xl:col-span-2 min-w-0">
   370→            <ConfluenceBoard confluences={data.confluences} focusSymbol={focusSymbol} now={now} />
   371→          </div>
   372→        </div>
   373→      </div>
   374→    </section>
   375→  );
   376→}

   530→function EventsBoard({ calendar, now }: { calendar: CalendarDigest; now: number }) {
   531→  const usdHigh = useMemo(
   532→    () => calendar.usdThisWeek.filter((e) => e.impact === "High"),
   533→    [calendar.usdThisWeek]
   534→  );
   535→  const nextHigh = useMemo(
   536→    () =>
   537→      [...calendar.nextHighImpact]
   538→        .filter((e) => e.ts >= now - 15 * 60_000)
   539→        .sort((a, b) => a.ts - b.ts)
   540→        .slice(0, 6),
   541→    [calendar.nextHighImpact, now]
   542→  );
   543→
   544→  return (
   545→    <section
   546→      className="rounded-lg border border-[#1e303d] bg-[#0d1720] p-3"
   547→      aria-label="Event risk and US economic calendar"

   658→const CONFLUENCE_FILTERS: Array<{ id: ConfluenceFilter; label: string }> = [

   666→function ConfluenceBoard({
   667→  confluences,
   668→  focusSymbol,
   669→  now,
   670→}: {
   671→  confluences: Confluence[];
   672→  focusSymbol: string;
   673→  now: number;
   674→}) {
   675→  const [filter, setFilter] = useState<ConfluenceFilter>("all");
   676→
   677→  const cards = useMemo(() => {
   678→    const passes = (c: Confluence) =>
   679→      filter === "all"
   680→        ? true
   681→        : filter === "strong"
   682→          ? c.strength === "STRONG"
   683→          : c.assetClass === filter;
   684→    const sorted = (list: Confluence[]) =>
   685→      [...list].sort(
   686→        (a, b) =>
   687→          (STRENGTH_RANK[a.strength] ?? 3) - (STRENGTH_RANK[b.strength] ?? 3) ||
   688→          Math.abs(b.trendScore) - Math.abs(a.trendScore)
   689→      );
   690→    const focused = confluences.filter((c) => c.symbol === focusSymbol).filter(passes);
   691→    const rest = confluences.filter((c) => c.symbol !== focusSymbol).filter(passes);
   692→    return [...sorted(focused), ...sorted(rest)];
   693→  }, [confluences, focusSymbol, filter]);
   694→
   695→  return (
   696→    <section className="rounded-lg border border-[#1e303d] bg-[#0d1720] p-3" aria-label="Confluence board">
   697→      <div className="flex flex-wrap items-center justify-between gap-2 mb-2">
   698→        <span className="text-[10px] font-extrabold tracking-[0.08em] text-[#aebdca]">
   699→          <Layers className="w-3 h-3 inline mr-1 -mt-0.5" />
   700→          CONFLUENCE BOARD · PER-MARKET ESTIMATES
   701→        </span>
   702→        <div className="flex items-center gap-1" role="group" aria-label="Confluence filter">
   703→          {CONFLUENCE_FILTERS.map((f) => (
   704→            <Button
   705→              key={f.id}
   706→              size="sm"
   707→              variant="ghost"
   708→              onClick={() => setFilter(f.id)}
   709→              aria-pressed={filter === f.id}
   710→              className={cn(
   711→                "h-6 px-2 text-[9px] gap-1 rounded-md",
   712→                filter === f.id
   713→                  ? "bg-[#203244] text-[#e5f0fb] border border-[#34506a]"
   714→                  : "text-[#8193a3] border border-transparent hover:text-[#c3d0dc]"
   715→              )}
   716→            >
   717→              {f.label}
   718→            </Button>
   719→          ))}
   720→        </div>
   721→      </div>
   722→      {cards.length === 0 ? (
   723→        <div className="text-[10px] text-[#687e90]">no markets match this filter</div>
   724→      ) : (
   725→        <div className="grid gap-2 grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 2xl:grid-cols-4">
   726→          {cards.map((c) => (
   727→            <ConfluenceCard key={c.symbol} c={c} focused={c.symbol === focusSymbol} now={now} />
   728→          ))}
   729→        </div>
   730→      )}
   731→    </section>
   732→  );
   733→}

   735→function TrendMeter({ score }: { score: number }) {
   736→  const clamped = Math.max(-3, Math.min(3, score));
   737→  const pct = ((clamped + 3) / 6) * 100;
   738→  const width = Math.abs(pct - 50);
   739→  const left = Math.min(pct, 50);
   740→  return (
   741→    <div className="flex items-center gap-2">
   742→      <div className="relative h-1.5 flex-1 rounded-full bg-[#16222d] overflow-hidden">
   743→        <div className="absolute inset-y-0 left-1/2 w-px bg-[#3a5268]" aria-hidden />
   744→        <div
   745→          className={cn("absolute inset-y-0 rounded-full", clamped >= 0 ? "bg-[#4bc39a]" : "bg-[#ef777d]")}
   746→          style={{ left: `${left}%`, width: `${width}%` }}
   747→          aria-hidden
   748→        />
   749→      </div>
   750→      <span
   751→        className={cn(
   752→          "text-[10px] font-bold tnum w-10 text-right",
   753→          clamped > 0.4 ? "text-[#4bc39a]" : clamped < -0.4 ? "text-[#ef777d]" : "text-[#8193a3]"
   754→        )}
   755→      >
   756→        {clamped > 0 ? "+" : ""}
   757→        {clamped.toFixed(1)}
   758→      </span>
   759→    </div>
   760→  );
   761→}
