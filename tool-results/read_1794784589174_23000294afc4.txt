   401→   401→async function requestJson<T>(url: string, method: "GET" | "POST" | "DELETE", body?: unknown): Promise<T> {
   402→   402→  const res = await fetch(url, {
   403→   403→    method,
   404→   404→    headers: body !== undefined ? { "Content-Type": "application/json" } : undefined,
   405→   405→    body: body !== undefined ? JSON.stringify(body) : undefined,
   406→   406→    cache: "no-store",
   407→   407→  });
   408→   408→  if (res.status === 401) throw new Error("Session expired — reload to sign in again.");
   409→   409→  if (!res.ok) {
   410→   410→    let msg = `HTTP ${res.status}`;
   411→   411→    try {
   412→   412→      const j = (await res.json()) as { error?: string };
   413→   413→      if (j?.error) msg = j.error;
   414→   414→    } catch {
   415→   415→      /* keep the status message */
   416→   416→    }
   417→   417→    throw new Error(msg);
   418→   418→  }
   419→   419→  return (await res.json()) as T;
   420→   420→}

   426→   426→/** Poll an endpoint until `isDone(data)` says the stream is terminal, then stop. */
   427→   427→function usePollUntil<T>(url: string | null, intervalMs: number, isDone: (data: T) => boolean) {
   428→   428→  const [data, setData] = useState<T | null>(null);
   429→   429→  const [loading, setLoading] = useState(false);
   430→   430→  const [error, setError] = useState<string | null>(null);
   431→   431→
   432→   432→  useEffect(() => {
   433→   433→    if (!url) return;
   434→   434→    const target = url;
   435→   435→    let alive = true;
   436→   436→    let timer: ReturnType<typeof setTimeout>;
   437→   437→    setData(null);
   438→   438→    setError(null);
   439→   439→    setLoading(true);
   440→   440→
   441→   441→    async function tick() {
   442→   442→      try {
   443→   443→        const res = await fetch(target, { cache: "no-store" });
   444→   444→        if (res.status === 401) throw new Error("Session expired — reload to sign in again.");
   445→   445→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   446→   446→        const json = (await res.json()) as T;
   447→   447→        if (!alive) return;
   448→   448→        setData(json);
   449→   449→        setError(null);
   450→   450→        if (isDone(json)) {
   451→   451→          setLoading(false);
   452→   452→          return;
   453→   453→        }
   454→   454→        timer = setTimeout(tick, intervalMs);
   455→   455→      } catch (e) {
   456→   456→        if (!alive) return;
   457→   457→        setError(e instanceof Error ? e.message : String(e));
   458→   458→        setLoading(false);
   459→   459→      }
   460→   460→    }
   461→   461→    tick();
   462→   462→    return () => {
   463→   463→      alive = false;
   464→   464→      clearTimeout(timer);
   465→   465→    };
   466→   466→  }, [url, intervalMs, isDone]);
   467→   467→
   468→   468→  return { data, loading, error };
   469→   469→}

   509→   509→  return parts.join(" + ") || "signal-only exit";
   510→   510→}

   534→   534→  const parts: string[] = [];
   535→   535→  for (const [k, v] of Object.entries(patch)) {
   536→   536→    if (v !== null && typeof v === "object") {
   537→   537→      for (const [k2, v2] of Object.entries(v as Record<string, unknown>)) {
   538→   538→        parts.push(`${k}.${k2}=${String(v2)}`);
   539→   539→      }
   540→   540→    } else {
   541→   541→      parts.push(`${k}=${String(v)}`);
   542→   542→    }
   543→   543→  }
   544→   544→  return parts.join(" ") || "copy";
   545→   545→}

   564→   564→  if (pts.length <= target) return pts;

   567→   567→  for (let i = 0; i < target; i++) out.push(pts[Math.floor(i * step)]);
