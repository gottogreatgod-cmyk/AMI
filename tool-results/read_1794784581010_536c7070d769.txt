   401→async function requestJson<T>(url: string, method: "GET" | "POST" | "DELETE", body?: unknown): Promise<T> {
   402→  const res = await fetch(url, {
   403→    method,
   404→    headers: body !== undefined ? { "Content-Type": "application/json" } : undefined,
   405→    body: body !== undefined ? JSON.stringify(body) : undefined,
   406→    cache: "no-store",
   407→  });
   408→  if (res.status === 401) throw new Error("Session expired — reload to sign in again.");
   409→  if (!res.ok) {
   410→    let msg = `HTTP ${res.status}`;
   411→    try {
   412→      const j = (await res.json()) as { error?: string };
   413→      if (j?.error) msg = j.error;
   414→    } catch {
   415→      /* keep the status message */
   416→    }
   417→    throw new Error(msg);
   418→  }
   419→  return (await res.json()) as T;
   420→}
   421→
   422→function postJson<T>(url: string, body: unknown): Promise<T> {
   423→  return requestJson<T>(url, "POST", body);
   424→}

   426→/** Poll an endpoint until `isDone(data)` says the stream is terminal, then stop. */
   427→function usePollUntil<T>(url: string | null, intervalMs: number, isDone: (data: T) => boolean) {
   428→  const [data, setData] = useState<T | null>(null);
   429→  const [loading, setLoading] = useState(false);
   430→  const [error, setError] = useState<string | null>(null);
   431→
   432→  useEffect(() => {
   433→    if (!url) return;
   434→    const target = url;
   435→    let alive = true;
   436→    let timer: ReturnType<typeof setTimeout>;
   437→    setData(null);
   438→    setError(null);
   439→    setLoading(true);
   440→
   441→    async function tick() {
   442→      try {
   443→        const res = await fetch(target, { cache: "no-store" });
   444→        if (res.status === 401) throw new Error("Session expired — reload to sign in again.");
   445→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   446→        const json = (await res.json()) as T;
   447→        if (!alive) return;
   448→        setData(json);
   449→        setError(null);
   450→        if (isDone(json)) {
   451→          setLoading(false);
   452→          return;
   453→        }
   454→        timer = setTimeout(tick, intervalMs);
   455→      } catch (e) {
   456→        if (!alive) return;
   457→        setError(e instanceof Error ? e.message : String(e));
   458→        setLoading(false);
   459→      }
   460→    }
   461→    tick();
   462→    return () => {
   463→      alive = false;
   464→      clearTimeout(timer);
   465→    };
   466→  }, [url, intervalMs, isDone]);
   467→
   468→  return { data, loading, error };
   469→}

   533→function patchSummary(patch: Record<string, unknown>): string {
   534→  const parts: string[] = [];
   535→  for (const [k, v] of Object.entries(patch)) {
   536→    if (v !== null && typeof v === "object") {
   537→      for (const [k2, v2] of Object.entries(v as Record<string, unknown>)) {
   538→        parts.push(`${k}.${k2}=${String(v2)}`);
   539→      }
   540→    } else {
   541→      parts.push(`${k}=${String(v)}`);
   542→    }
   543→  }
   544→  return parts.join(" ") || "copy";
   545→}

   563→function downsampleEquity(pts: EquityPoint[], target: number): EquityPoint[] {
   564→  if (pts.length <= target) return pts;
   565→  const step = pts.length / target;
   566→  const out: EquityPoint[] = [];
   567→  for (let i = 0; i < target; i++) out.push(pts[Math.floor(i * step)]);
   568→  out[out.length - 1] = pts[pts.length - 1];
   569→  return out;
   570→}

   576→  return next;
